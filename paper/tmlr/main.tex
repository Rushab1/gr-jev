% The tables are sized for the ACL column of 7.7 cm. In the TMLR column of 16.5 cm, a table stretched to the column
% width by \resizebox would grow larger than the text, and one set in \footnotesize would be small. Here a table only
% shrinks to fit the column, and every table is set in \small. A table* spans both ACL columns, about the TMLR
% width, and gets narrower space between columns to fit at that size.
\usepackage{adjustbox}
\usepackage{etoolbox}
\renewcommand{\resizebox}[3]{\adjustbox{max width=#1}{#3}}
\AtBeginEnvironment{table}{\small\let\footnotesize\small\let\scriptsize\small}
\AtBeginEnvironment{table*}{\small\let\footnotesize\small\let\scriptsize\small\setlength{\tabcolsep}{4pt}}
